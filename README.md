    const students=[
      {id:"001",activity:28,exercise:27,test:37},
      {id:"002",activity:25,exercise:24,test:31},
      {id:"003",activity:30,exercise:29,test:39},
      {id:"004",activity:23,exercise:22,test:27},
      {id:"005",activity:27,exercise:26,test:34},
      {id:"006",activity:24,exercise:23,test:29},
      {id:"007",activity:29,exercise:28,test:36},
      {id:"008",activity:26,exercise:25,test:31}
    ];
    const total=s=>s.activity+s.exercise+s.test;
    const grade=n=>n>=80?"ดีเยี่ยม":n>=70?"ดี":"ผ่านเกณฑ์";
    const tbody=document.getElementById("score-rows");
    const search=document.getElementById("search-id");
    const filter=document.getElementById("grade-filter");
    document.getElementById("average-score").innerHTML=(students.reduce((sum,s)=>sum+total(s),0)/students.length).toFixed(1)+' <small>คะแนน</small>';
    document.getElementById("student-count").innerHTML=students.length+' <small>คน</small>';
    function render(){
      const query=search.value.trim();
      const selected=filter.value;
      const list=students.filter(s=>s.id.includes(query)&&(selected==="all"||grade(total(s))===selected));
      tbody.innerHTML=list.map(s=>{
        const score=total(s),level=grade(score),badge=level==="ดีเยี่ยม"?"":level==="ดี"?" good":" fair";
        return `<tr><td>ป.1/6-${s.id}</td><td>${s.activity}</td><td>${s.exercise}</td><td>${s.test}</td><td class="total">${score}</td><td><span class="badge${badge}">${level}</span></td></tr>`;
      }).join("");
      document.getElementById("result-count").textContent=`แสดง ${list.length} รายการ`;
      document.getElementById("empty-state").style.display=list.length?"none":"block";
    }
    search.addEventListener("input",render);
    filter.addEventListener("change",render);
    document.getElementById("reset-filters").addEventListener("click",()=>{search.value="";filter.value="all";render();search.focus()});
    render();
    lucide.createIcons();
  </script>
